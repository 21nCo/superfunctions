# Phase 01: Core Drizzle Schema Codegen

**Goal**: Implement `generateDrizzleSchema()` function that produces valid Drizzle table definitions for all 3 dialects.
**Duration Estimate**: 3-4 hours

## In Scope

### Features
- `generateDrizzleSchema(schema: DatafnSchema, dialect: "postgres" | "mysql" | "sqlite"): string`
- Type mapping for all 7 DataFn field types across 3 dialects
- Field modifiers: `.notNull()`, `.unique()`, `.default()`, `.primaryKey()`
- camelCase → snake_case column name conversion

- Join table generation from `many-many` relations
- Metadata fields in join tables
- Custom `joinTable` and `joinColumns` support
- Schema export object (`export const schema = { ... }`)
- Generated file header comment
- Deterministic output (sorted resources, fields, imports)
- Import tree-shaking (only import used column types)

## Deliverables

### New File: `datafn/cli/src/drizzle-codegen.ts`

```typescript
import { type DatafnSchema, validateSchema, unwrapEnvelope } from "@datafn/core";

type Dialect = "postgres" | "mysql" | "sqlite";

/**
 * Generate Drizzle ORM schema from a DataFn schema.
 */
export function generateDrizzleSchema(schemaInput: unknown, dialect: Dialect): string {
  const schema = unwrapEnvelope(validateSchema(schemaInput));
  const lines: string[] = [];
  const usedImports = new Set<string>();
  const tableConstructor = getTableConstructor(dialect);
  const importPath = getImportPath(dialect);

  // Track all table exports
  const tableExports: string[] = [];

  // Header
  lines.push("// Generated by @datafn/cli. DO NOT EDIT.");
  lines.push(`// Database: ${dialect}`);
  lines.push("");

  // Sort resources alphabetically
  const resources = [...schema.resources].sort((a, b) => a.name.localeCompare(b.name));

  // Generate resource tables
  for (const resource of resources) {
    const tableName = resource.name;
    const varName = sanitizeVarName(tableName);
    tableExports.push(varName);
    usedImports.add(tableConstructor);

    const fields = [...resource.fields].sort((a, b) => a.name.localeCompare(b.name));
    const hasIndices = resource.indices &&
      typeof resource.indices === "object" &&
      !Array.isArray(resource.indices) &&
      resource.indices.base?.length;

    lines.push(`export const ${varName} = ${tableConstructor}("${tableName}", {`);

    for (const field of fields) {
      const colDef = buildColumnDefinition(field, dialect, usedImports);
      lines.push(`  ${field.name}: ${colDef},`);
    }

    if (hasIndices) {
      const baseIndices = (resource.indices as { base?: string[] }).base!;
      usedImports.add("index");
      lines.push(`}, (table) => [`);
      for (const idxField of baseIndices) {
        lines.push(`  index("${tableName}_${camelToSnake(idxField)}_idx").on(table.${idxField}),`);
      }
      lines.push(`]);`);
    } else {
      lines.push(`});`);
    }
    lines.push("");
  }

  // Generate join tables for many-many relations
  const relations = schema.relations ?? [];
  for (const rel of [...relations].sort((a, b) => {
    const aName = getJoinTableName(rel);
    const bName = getJoinTableName(rel);
    return aName.localeCompare(bName);
  })) {
    if (rel.type !== "many-many") continue;

    const jtName = getJoinTableName(rel);
    const varName = sanitizeVarName(jtName);
    tableExports.push(varName);
    usedImports.add(tableConstructor);

    const fromCol = rel.joinColumns?.from ?? "from";
    const toCol = rel.joinColumns?.to ?? "to";

    lines.push(`export const ${varName} = ${tableConstructor}("${jtName}", {`);

    // Build join columns + metadata, sorted
    const joinFields: Array<{ name: string; def: string }> = [
      { name: fromCol, def: buildSimpleColumn(fromCol, "string", dialect, usedImports, true) },
      { name: "id", def: buildSimpleColumn("id", "string", dialect, usedImports, false, true) },
      { name: toCol, def: buildSimpleColumn(toCol, "string", dialect, usedImports, true) },
    ];

    if (rel.metadata) {
      for (const meta of rel.metadata) {
        joinFields.push({
          name: meta.name,
          def: buildSimpleColumn(meta.name, meta.type, dialect, usedImports, false),
        });
      }
    }

    joinFields.sort((a, b) => a.name.localeCompare(b.name));

    for (const jf of joinFields) {
      lines.push(`  ${jf.name}: ${jf.def},`);
    }

    lines.push(`});`);
    lines.push("");
  }

  // Schema export
  tableExports.sort();
  lines.push(`export const schema = { ${tableExports.join(", ")} };`);
  lines.push("");

  // Build import line
  const sortedImports = [...usedImports].sort();
  const importLine = `import { ${sortedImports.join(", ")} } from "${importPath}";`;

  // Insert import after header
  lines.splice(2, 0, importLine);

  return lines.join("\n");
}
```

### Helper Functions

```typescript
function getTableConstructor(dialect: Dialect): string {
  switch (dialect) {
    case "postgres": return "pgTable";
    case "mysql": return "mysqlTable";
    case "sqlite": return "sqliteTable";
  }
}

function getImportPath(dialect: Dialect): string {
  switch (dialect) {
    case "postgres": return "drizzle-orm/pg-core";
    case "mysql": return "drizzle-orm/mysql-core";
    case "sqlite": return "drizzle-orm/sqlite-core";
  }
}

function camelToSnake(str: string): string {
  return str.replace(/[A-Z]/g, (letter) => `_${letter.toLowerCase()}`);
}

function sanitizeVarName(name: string): string {
  // Ensure valid JS identifier (handles __datafn_join_* names)
  return name.replace(/-/g, "_");
}

function buildColumnDefinition(
  field: DatafnFieldSchema,
  dialect: Dialect,
  usedImports: Set<string>,
): string {
  const snakeName = camelToSnake(field.name);
  const { call, importName } = getColumnType(field.type, dialect, snakeName);
  usedImports.add(importName);

  let def = call;

  if (field.name === "id") {
    def += ".primaryKey()";
  } else {
    if (field.required) def += ".notNull()";
    if (field.unique === true) def += ".unique()";
  }

  if (field.default !== undefined) {
    const defaultVal = typeof field.default === "string"
      ? `"${field.default}"`
      : String(field.default);
    def += `.default(${defaultVal})`;
  }

  return def;
}
```

### Type Mapping Implementation

```typescript
function getColumnType(
  fieldType: string,
  dialect: Dialect,
  sqlName: string,
): { call: string; importName: string } {
  if (dialect === "postgres") {
    switch (fieldType) {
      case "string":
      case "file":
        return { call: `text("${sqlName}")`, importName: "text" };
      case "number":
        return { call: `numeric("${sqlName}")`, importName: "numeric" };
      case "boolean":
        return { call: `boolean("${sqlName}")`, importName: "boolean" };
      case "date":
        return { call: `bigint("${sqlName}", { mode: "number" })`, importName: "bigint" };
      case "object":
      case "array":
        return { call: `jsonb("${sqlName}")`, importName: "jsonb" };
      default:
        return { call: `text("${sqlName}")`, importName: "text" };
    }
  }
  if (dialect === "mysql") {
    switch (fieldType) {
      case "string":
      case "file":
        return { call: `text("${sqlName}")`, importName: "text" };
      case "number":
        return { call: `int("${sqlName}")`, importName: "int" };
      case "boolean":
        return { call: `boolean("${sqlName}")`, importName: "boolean" };
      case "date":
        return { call: `bigint("${sqlName}", { mode: "number" })`, importName: "bigint" };
      case "object":
      case "array":
        return { call: `json("${sqlName}")`, importName: "json" };
      default:
        return { call: `text("${sqlName}")`, importName: "text" };
    }
  }
  // sqlite
  switch (fieldType) {
    case "string":
    case "file":
    case "object":
    case "array":
      return { call: `text("${sqlName}")`, importName: "text" };
    case "number":
    case "boolean":
    case "date":
      return { call: `integer("${sqlName}")`, importName: "integer" };
    default:
      return { call: `text("${sqlName}")`, importName: "text" };
  }
}
```
